seed_assimilation_and_conversion = {
	save_scope_as = new_state
	owner = {
		random_primary_culture = {
			save_scope_as = spawning_culture
		}
	}
	if = {
		limit = {
			NOT = {
				has_cultural_community = scope:spawning_culture
			}
		}
		ordered_scope_pop = {
			limit = {
				OR = {
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = european_heritage 
						}
						culture = {
							has_discrimination_trait = european_heritage 
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = middle_eastern_heritage  
						}	
						culture = {
							has_discrimination_trait = middle_eastern_heritage  
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = central_asian_heritage  
						}
						culture = {
							has_discrimination_trait = central_asian_heritage  
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = african_heritage  
						}
						culture = {
							has_discrimination_trait = african_heritage  
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = south_asian_heritage  
						}
						culture = {
							has_discrimination_trait = south_asian_heritage  
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = southeast_asian_heritage  
						}
						culture = {
							has_discrimination_trait = southeast_asian_heritage  
						}
					}
					AND = {
						scope:spawning_culture = {
							has_discrimination_trait = east_asian_heritage  
						}
						culture = {
							has_discrimination_trait = east_asian_heritage  
						}
					}
					AND = {
						#scope:spawning_culture = {
						#	has_discrimination_trait = indigenous_american_heritage 
						#}
						culture = {
							has_discrimination_trait = indigenous_american_heritage 
						}
					}
					AND = {
						#scope:spawning_culture = {
						#	has_discrimination_trait = indigenous_oceanic_heritage  
						#}
						culture = {
							has_discrimination_trait = indigenous_oceanic_heritage  
						}
					}
				}
			}
			order_by = total_size
			min=0
			max=10
			check_range_bounds = no
			change_pop_culture = { 
				target = scope:spawning_culture 
				value = 0.0001
			}
		}
	}
	convert_population = {
		target = owner.religion
		value = 0.0001
	}
	
	#### CUSTOMS
	if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = francophone } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_antillean
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = hispanophone } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_caribeno
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = anglophone } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_caribbean
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = italophone } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_italian
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = german_speaking } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_german
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = beneluxian_culture_group } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_dutch
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = scandinavian_culture_group } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_scandinavian
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = west_slavic } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_slavic
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = east_slavic } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_slavic
				value = 0.0001
			}
		}
	}
	else_if = {
		limit = { owner = { any_primary_culture = { has_discrimination_trait = lusophone } } }
		every_scope_pop = {
			limit = { culture = { has_discrimination_trait = african_heritage } }
			change_pop_culture = { 
				target = cu:afro_brazilian
				value = 0.0001
			}
		}
	}
}
